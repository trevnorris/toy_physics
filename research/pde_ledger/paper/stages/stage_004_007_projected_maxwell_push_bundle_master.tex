\label{eq:stage004003-projected-shifts}

\end{equation}

\[
D_0=K-B_0-Z_0,\qquad D_2=-(M+B_2+Z_2),\qquad D_4=-(B_4+Z_4),
\]
the first response variations are
\begin{equation}
\label{eq:stage004007-du2}
\delta u_2=\frac{D_0z_2-D_2z_0}{D_0^2},
\end{equation}
\begin{equation}
\label{eq:stage004007-du4}
\delta u_4=
\frac{D_0^2z_4-D_0(2D_2z_2+D_4z_0)+2D_2^2z_0}{D_0^3},
\end{equation}
and
\begin{equation}
\label{eq:stage004007-dp0}
\delta P_0=\frac{D_0n_0+N_0z_0}{D_0^2}.
\end{equation}
