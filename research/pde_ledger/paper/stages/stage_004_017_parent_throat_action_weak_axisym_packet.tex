\subsection*{Stage 004.017: Parent throat action weak-axisymmetric packet}
\label{subsec:stage004017-parent-throat-action-weak-axisym-packet}

\paragraph{Source and audit.}
This substage corresponds to
\StageFile{scripts/moving_throat_pde_stage004_017_parent_throat_action_weak_axisym_packet_sympy_audit.py}.
It solves the first-order even gates in parent-wall slope variables.

\paragraph{Slope packet.}
Let
\begin{equation}
\label{eq:stage004017-wall-slopes}
\delta M_\Sigma=\int dw\,\delta\mu_\eta\,\beta_2^2,\qquad
\delta K_\Sigma=\int dw\,\Bigl[\delta T_w(\beta_2')^2+
(\delta K_\eta+6\delta T_\Omega)\beta_2^2\Bigr].
\end{equation}
Then
\begin{equation}
\label{eq:stage004017-denominator-slopes}
D_{01}=\delta K_\Sigma-B_{01}-Z_{01},\quad
D_{21}=-(\delta M_\Sigma+B_{21}+Z_{21}),\quad
D_{41}=-(B_{41}+Z_{41}).
\end{equation}
The live gates are
\begin{equation}
\label{eq:stage004017-gates}
K_1=D_{21}+\frac{D_{01}}9,\qquad
H_{\rm even}=D_{41}-\frac23D_{21}-\frac{D_{01}}{27},
\end{equation}
\begin{equation}
\label{eq:stage004017-xi}
\Xi_1=\frac{N_{01}}{N_0}-\frac{D_{01}}{D_0}.
\end{equation}

\paragraph{Even compensation.}
The exact wall-slope solve is
\begin{equation}
\label{eq:stage004017-wall-slope-solve}
\delta K_\Sigma=B_{01}+Z_{01}+27(B_{41}+Z_{41}),
\qquad
\delta M_\Sigma=-(B_{21}+Z_{21})+3(B_{41}+Z_{41}).
\end{equation}
After this solve, the residual prefactor obstruction is
\begin{equation}
\label{eq:stage004017-residual-xi}
\Xi_1=\frac{N_{01}}{N_0}
-\frac{27(B_{41}+Z_{41})}{K_\Sigma-B_0-Z_0}.
\end{equation}

\paragraph{Output.}
Stage~004.017 exports the weak-axisymmetric parent packet
\eqref{eq:stage004017-wall-slopes}--\eqref{eq:stage004017-residual-xi}.
