\subsection*{Derivation}

\paragraph{1. Weighted grouped projector calculus.}
The grouped triple \((20,21,22)\) carries metric
\begin{equation}
\label{eq:app-stage006-grouped-metric}
G_{\rm grp}=\operatorname{diag}(1,2,2).
\end{equation}
Use the \(G_{\rm grp}\)-orthogonal basis
\begin{equation}
\label{eq:app-stage006-basis}
e_{\bar x}=(1,1,1),
\qquad
e_a=(4,-1,-1),
\qquad
e_b=(0,1,-1).
\end{equation}
The norms are
\begin{equation}
\label{eq:app-stage006-norms}
e_{\bar x}^TG_{\rm grp}e_{\bar x}=5,
\qquad
e_a^TG_{\rm grp}e_a=20,
\qquad
e_b^TG_{\rm grp}e_b=4.
\end{equation}
Thus any grouped vector decomposes as
\begin{equation}
\label{eq:app-stage006-decomp}
x=\bar x\,e_{\bar x}+a_xe_a+b_xe_b,
\end{equation}
with \(\bar x,a_x,b_x\) as in \eqref{eq:app-stage005-grouped-defs}.  The exact projectors are
\begin{equation}
\label{eq:app-stage006-projectors}
\boxed{
P_{\bar x}=\frac15
\begin{pmatrix}
1&2&2\\1&2&2\\1&2&2
\end{pmatrix},
\quad
P_a=\frac1{20}
\begin{pmatrix}
16&-8&-8\\-4&2&2\\-4&2&2
\end{pmatrix},
\quad
P_b=\frac14
\begin{pmatrix}
0&0&0\\0&2&-2\\0&-2&2
\end{pmatrix}}.
\end{equation}
They obey
\begin{equation}
\label{eq:app-stage006-projector-identities}
P_{\bar x}+P_a+P_b=I_3,
\qquad
P_iP_j=\delta_{ij}P_i.
\end{equation}
Grouped isotropy is the condition
\begin{equation}
\label{eq:app-stage006-isotropy-projector}
P_ax=P_bx=0.
\end{equation}

\paragraph{2. Full coupled lane Lagrangian.}
For each lane \(A\in\{20,21,22\}\), retain a wall amplitude \(q_A\), stable BdG support modes \(X_{A\alpha}\), brane-like gauge coordinates \(U_{A r}\), mixed coordinates \(W_{A r}\), and internal gauge mixing \(R_{A r}\).  The reduced Lagrangian is
\begin{align}
\label{eq:app-stage006-full-lagrangian}
L_A={}&\frac12M_A\dot q_A^2-\frac12K_Aq_A^2
+\sum_\alpha\left[\frac12\dot X_{A\alpha}^2-\frac12\varpi_{A\alpha}^2X_{A\alpha}^2+c_{A\alpha}q_AX_{A\alpha}\right]\\
&+\sum_r\bigl[\frac12\dot U_{A r}^2-\frac12\Omega_{U,A r}^2U_{A r}^2
+\frac12\dot W_{A r}^2-\frac12\Omega_{W,A r}^2W_{A r}^2
+R_{A r}U_{A r}W_{A r}\\
&\qquad\quad
+g_{U,A r}q_AU_{A r}+g_{W,A r}q_AW_{A r}\bigr].
\end{align}

\paragraph{3. BdG support moments.}
Define
\begin{equation}
\label{eq:app-stage006-b-moments}
\boxed{
B_{A0}=\sum_\alpha\frac{c_{A\alpha}^2}{\varpi_{A\alpha}^2},
\qquad
B_{A2}=\sum_\alpha\frac{c_{A\alpha}^2}{\varpi_{A\alpha}^4},
\qquad
B_{A4}=\sum_\alpha\frac{c_{A\alpha}^2}{\varpi_{A\alpha}^6}}.
\end{equation}
These are the Stage~003 Schur-complement moments collected lane by lane.

\paragraph{4. Conservative Maxwell/mixed moments.}
For each port \(r\), set
\begin{equation}
\label{eq:app-stage006-delta-s-q-g}
\Delta_{A r}=\Omega_{U,A r}^2\Omega_{W,A r}^2-R_{A r}^2,
\qquad
S_{A r}=\Omega_{U,A r}^2+\Omega_{W,A r}^2,
\end{equation}
\begin{equation}
Q_{A r}=g_{U,A r}^2\Omega_{W,A r}^2
+2g_{U,A r}g_{W,A r}R_{A r}
+g_{W,A r}^2\Omega_{U,A r}^2,
\qquad
G_{A r}=g_{U,A r}^2+g_{W,A r}^2.
\end{equation}
The conservative Maxwell/mixed moments are
\begin{align}
\label{eq:app-stage006-z-moments-port}
Z_{A0}^{(r)}&=\frac{Q_{A r}}{\Delta_{A r}},\\
Z_{A2}^{(r)}&=\frac{Q_{A r}S_{A r}-G_{A r}\Delta_{A r}}{\Delta_{A r}^2},\\
Z_{A4}^{(r)}&=\frac{Q_{A r}(S_{A r}^2-\Delta_{A r})-S_{A r}G_{A r}\Delta_{A r}}{\Delta_{A r}^3},
\end{align}
and
\begin{equation}
\label{eq:app-stage006-z-sum}
Z_{An}=\sum_rZ_{An}^{(r)},
\qquad
n\in\{0,2,4\}.
\end{equation}

\paragraph{5. Outgoing-transfer moments.}
Define
\begin{equation}
\label{eq:app-stage006-p-ar}
P_{A r}=\Omega_{U,A r}^2g_{W,A r}+R_{A r}g_{U,A r}.
\end{equation}
The outgoing-transfer moments per port are
\begin{align}
\label{eq:app-stage006-n-moments-port}
N_{A0}^{(r)}&=\frac{P_{A r}^2}{\Delta_{A r}^2},\\
N_{A2}^{(r)}&=\frac{2P_{A r}\bigl(P_{A r}S_{A r}-\Delta_{A r}g_{W,A r}\bigr)}{\Delta_{A r}^3},\\
N_{A4}^{(r)}&=\frac{\Delta_{A r}^2g_{W,A r}^2-2\Delta_{A r}P_{A r}^2-4\Delta_{A r}P_{A r}S_{A r}g_{W,A r}+3P_{A r}^2S_{A r}^2}{\Delta_{A r}^4}.
\end{align}
Summing over outgoing ports gives
\begin{equation}
\label{eq:app-stage006-n-sum}
N_{An}=\sum_rN_{An}^{(r)},
\qquad
n\in\{0,2,4\}.
\end{equation}

\paragraph{6. Total conservative grouped operator.}
The full conservative grouped-lane operator is
\begin{equation}
\label{eq:app-stage006-d-total}
D_A^{\rm cons}(\omega)=D_{A0}+D_{A2}\omega^2+D_{A4}\omega^4+O(\omega^6),
\end{equation}
with
\begin{equation}
\label{eq:app-stage006-d-coeffs}
\boxed{
D_{A0}=K_A-B_{A0}-Z_{A0},
\qquad
D_{A2}=-\bigl(M_A+B_{A2}+Z_{A2}\bigr),
\qquad
D_{A4}=-\bigl(B_{A4}+Z_{A4}\bigr)}.
\end{equation}
This is the full Part-I coefficient ledger.

\paragraph{7. Microscopic anisotropy ledger.}
Apply the grouped decomposition \eqref{eq:app-stage006-decomp} to every microscopic family \(K_A,M_A,B_{An},Z_{An},N_{An}\).  The static conservative defects obey
\begin{equation}
\label{eq:app-stage006-d0-defects}
\bar D_0=\bar K-\bar B_0-\bar Z_0,
\qquad
a_{D,0}=a_K-a_{B,0}-a_{Z,0},
\qquad
b_{D,0}=b_K-b_{B,0}-b_{Z,0}.
\end{equation}
Similarly,
\begin{equation}
\label{eq:app-stage006-d2-defects}
\bar D_2=-(\bar M+\bar B_2+\bar Z_2),
\qquad
a_{D,2}=-(a_M+a_{B,2}+a_{Z,2}),
\qquad
b_{D,2}=-(b_M+b_{B,2}+b_{Z,2}),
\end{equation}
and
\begin{equation}
\label{eq:app-stage006-d4-defects}
\bar D_4=-(\bar B_4+\bar Z_4),
\qquad
a_{D,4}=-(a_{B,4}+a_{Z,4}),
\qquad
b_{D,4}=-(b_{B,4}+b_{Z,4}).
\end{equation}
Thus grouped anisotropy can enter only through the wall baseline, the BdG support bundle, or the Maxwell/mixed conservative and outgoing-transfer bundles.

\paragraph{8. Isotropic branch and normalization ratio.}
On the isotropic branch,
\begin{equation}
D_{20,n}=D_{21,n}=D_{22,n}=D_n,
\qquad
N_{20,n}=N_{21,n}=N_{22,n}=N_n.
\end{equation}
Stage~005 then gives
\begin{equation}
\label{eq:app-stage006-isotropic-response}
u_2=-\frac{D_2}{D_0},
\qquad
u_4=\frac{D_2^2-D_0D_4}{D_0^2},
\end{equation}
and
\begin{equation}
\label{eq:app-stage006-isotropic-pref}
P_0=\frac{N_0}{D_0},
\qquad
P_2=\frac{D_0N_2-2D_2N_0}{D_0^2},
\end{equation}
\begin{equation}
\label{eq:app-stage006-isotropic-p4}
P_4=\frac{D_0^2N_4-2D_0(D_2N_2+D_4N_0)+3D_2^2N_0}{D_0^3}.
\end{equation}
The isotropic normalization test is
\begin{equation}
\label{eq:app-stage006-normalization-test}
\boxed{
\widehat m_0^{\,2}\frac{N_0}{K-B_0-Z_0}
=\frac{54Gc_s^5}{5a^5c^5}}.
\end{equation}
The denominator is the static full-bundle stiffness
\begin{equation}
D_0=K-B_0-Z_0.
\end{equation}

\paragraph{9. Constant-prefactor branch.}
The constant-prefactor branch through the retained orders is
\begin{equation}
P_2=0,
\qquad
P_4=0.
\end{equation}
Using \eqref{eq:app-stage006-isotropic-pref}--\eqref{eq:app-stage006-isotropic-p4}, this is equivalent to
\begin{equation}
\label{eq:app-stage006-constant-prefactor-conditions}
\boxed{
N_2=\frac{2D_2N_0}{D_0},
\qquad
N_4=\frac{2D_0(D_2N_2+D_4N_0)-3D_2^2N_0}{D_0^2}}.
\end{equation}
So constant outgoing prefactor does not require \(N_2=N_4=0\); it requires precise correlation between transfer and conservative moments.

\paragraph{10. First-order anisotropy transport.}
Let
\begin{equation}
D_{An}=D_n+\delta D_{An},
\qquad
N_{An}=N_n+\delta N_{An},
\end{equation}
near the isotropic branch.  From \(u_2^{(A)}=-D_{A2}/D_{A0}\),
\begin{equation}
\label{eq:app-stage006-du2}
\delta u_2^{(A)}=-\frac{\delta D_{A2}+u_2\delta D_{A0}}{D_0}.
\end{equation}
Therefore
\begin{equation}
\label{eq:app-stage006-u-anisotropy}
\boxed{
a_{u,2}=-\frac{a_{D,2}+u_2a_{D,0}}{D_0},
\qquad
b_{u,2}=-\frac{b_{D,2}+u_2b_{D,0}}{D_0}}.
\end{equation}
Similarly, from \(P_0^{(A)}=N_{A0}/D_{A0}\),
\begin{equation}
\label{eq:app-stage006-dp0}
\delta P_0^{(A)}=\frac{\delta N_{A0}-P_0\delta D_{A0}}{D_0},
\end{equation}
so
\begin{equation}
\label{eq:app-stage006-p-anisotropy}
\boxed{
a_{P,0}=\frac{a_{N,0}-P_0a_{D,0}}{D_0},
\qquad
b_{P,0}=\frac{b_{N,0}-P_0b_{D,0}}{D_0}}.
\end{equation}
Thus outgoing-prefactor anisotropy can arise only from direct anisotropy in \(N_{A0}\) or conservative anisotropy in \(D_{A0}\) reweighted by \(P_0\).
