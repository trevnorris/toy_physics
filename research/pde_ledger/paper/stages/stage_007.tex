\subsection*{Derivation}

\paragraph{1. Real-STF harmonic normalization.}
Let \(n^i\) be the unit vector on the throat sphere and let \(E_A^{ij}\) be a real STF basis satisfying
\begin{equation}
\label{eq:app-stage007-STF-norm}
\operatorname{Tr}(E_AE_B)=\delta_{AB}.
\end{equation}
Define
\begin{equation}
\label{eq:app-stage007-YA}
Y_A(n)=\sqrt{\frac{15}{8\pi}}E_A^{ij}n_in_j,
\qquad
A\in\{20,21c,21s,22c,22s\}.
\end{equation}
The exact fourth-moment identity
\begin{equation}
\label{eq:app-stage007-fourth-moment}
\int_{S^2}n_in_jn_kn_l\,d\Omega
=\frac{4\pi}{15}\left(\delta_{ij}\delta_{kl}+\delta_{ik}\delta_{jl}+\delta_{il}\delta_{jk}\right)
\end{equation}
then gives
\begin{equation}
\label{eq:app-stage007-Y-orthonormal}
\boxed{
\int_{S^2}Y_A Y_B\,d\Omega=\delta_{AB}}.
\end{equation}
Thus the real grouped quadrupole ports have a canonical normalized angular basis.

\paragraph{2. Angular source-map identity.}
Expand the orbital/worldtube STF quadrupole source as
\begin{equation}
\label{eq:app-stage007-source-expansion}
S(n)=\sum_A S_A Y_A(n).
\end{equation}
Projecting on the same ports gives
\begin{equation}
\label{eq:app-stage007-source-map}
\boxed{
S_A^{\rm port}=\int_{S^2}Y_A(n)S(n)\,d\Omega=S_A}.
\end{equation}
Therefore the source map factors as
\begin{equation}
\label{eq:app-stage007-mhat-factor}
\widehat m_0=\widehat m_{\rm rad}\widehat m_{\rm ang},
\qquad
\boxed{\widehat m_{\rm ang}=1}
\end{equation}
on the canonical isotropic branch.  The remaining source-map problem is radial/axial.

\paragraph{3. Isotropic overlap factorization.}
On an \(O(3)\)-invariant branch, write the wall, BdG, brane-like gauge, and mixed coordinates in separated form:
\begin{align}
\eta_A(s,\Omega,t)&=q_A(t)\chi_\eta(s)Y_A(\Omega),\label{eq:app-stage007-eta-sep}\\
X_{\alpha,A}(s,\Omega,t)&=X_{\alpha,A}(t)\phi_\alpha(s)Y_A(\Omega),\label{eq:app-stage007-X-sep}\\
U_{r,A}(s,\Omega,t)&=U_{r,A}(t)u_r(s)Y_A(\Omega),\label{eq:app-stage007-U-sep}\\
W_{r,A}(s,\Omega,t)&=W_{r,A}(t)w_r(s)Y_A(\Omega).\label{eq:app-stage007-W-sep}
\end{align}
The angular factor is diagonal by \eqref{eq:app-stage007-Y-orthonormal}.  The radial/axial overlaps are
\begin{align}
I_{\eta\alpha}&=\int ds\,\mu_s(s)\chi_\eta(s)\phi_\alpha(s),\label{eq:app-stage007-I-eta-alpha}\\
I_{\eta u,r}&=\int ds\,\mu_s(s)\chi_\eta(s)u_r(s),\label{eq:app-stage007-I-eta-u}\\
I_{\eta w,r}&=\int ds\,\mu_s(s)\chi_\eta(s)w_r(s),\label{eq:app-stage007-I-eta-w}\\
I_{uw,r}&=\int ds\,\mu_s(s)u_r(s)w_r(s).\label{eq:app-stage007-I-uw}
\end{align}
The lane-independent reduced couplings are
\begin{equation}
\label{eq:app-stage007-isotropic-couplings}
\boxed{
C_\alpha=\lambda_{B,\alpha}I_{\eta\alpha},
\qquad
G_{U,r}=\lambda_{U,r}I_{\eta u,r},
\qquad
G_{W,r}=\lambda_{W,r}I_{\eta w,r},
\qquad
R_r=\lambda_{R,r}I_{uw,r}}.
\end{equation}
Thus
\begin{equation}
\label{eq:app-stage007-B-moments}
\boxed{
B_0=\sum_\alpha\frac{C_\alpha^2}{\varpi_\alpha^2},
\qquad
B_2=\sum_\alpha\frac{C_\alpha^2}{\varpi_\alpha^4},
\qquad
B_4=\sum_\alpha\frac{C_\alpha^2}{\varpi_\alpha^6}}.
\end{equation}
For each Maxwell/mixed pair define
\begin{equation}
\label{eq:app-stage007-Delta-S-Q-H-P}
\Delta_r=\Omega_{U,r}^2\Omega_{W,r}^2-R_r^2,
\qquad
S_r=\Omega_{U,r}^2+\Omega_{W,r}^2,
\end{equation}
\begin{equation}
Q_r=G_{U,r}^2\Omega_{W,r}^2+2G_{U,r}G_{W,r}R_r+G_{W,r}^2\Omega_{U,r}^2,
\qquad
H_r=G_{U,r}^2+G_{W,r}^2,
\end{equation}
\begin{equation}
\mathcal P_r=\Omega_{U,r}^2G_{W,r}+R_rG_{U,r}.
\end{equation}
Then
\begin{align}
\label{eq:app-stage007-ZN-moments}
Z_0&=\sum_r\frac{Q_r}{\Delta_r},&
Z_2&=\sum_r\frac{Q_rS_r-H_r\Delta_r}{\Delta_r^2},&
Z_4&=\sum_r\frac{Q_r(S_r^2-\Delta_r)-S_rH_r\Delta_r}{\Delta_r^3},\\
N_0&=\sum_r\frac{\mathcal P_r^2}{\Delta_r^2},&
N_2&=\sum_r\frac{2\mathcal P_r(\mathcal P_rS_r-\Delta_rG_{W,r})}{\Delta_r^3},&
N_4&=\sum_r\frac{\Delta_r^2G_{W,r}^2-2\Delta_r\mathcal P_r^2-4\Delta_r\mathcal P_rS_rG_{W,r}+3\mathcal P_r^2S_r^2}{\Delta_r^4}.
\end{align}
The conservative operator is
\begin{equation}
\label{eq:app-stage007-D-isotropic}
\boxed{
\begin{aligned}
D(\omega)&=D_0+D_2\omega^2+D_4\omega^4+O(\omega^6),\\
D_0&=K-B_0-Z_0,\\
D_2&=-(M+B_2+Z_2),\\
D_4&=-(B_4+Z_4).
\end{aligned}
}
\end{equation}
Therefore
\begin{equation}
\label{eq:app-stage007-isotropic-collapse}
\boxed{
\begin{aligned}
D_{20,n}=D_{21,n}=D_{22,n}&=D_n,\\
N_{20,n}=N_{21,n}=N_{22,n}&=N_n,
\end{aligned}
}
\end{equation}
so every grouped anisotropy defect vanishes on a truly \(O(3)\)-invariant branch.

\paragraph{4. Weak-axisymmetric quadrupole splitting.}
Let the first symmetry-breaking perturbation be
\begin{equation}
\label{eq:app-stage007-deltaK-axisym}
\delta K=\varepsilon\,\kappa(s)Y_{20}(\Omega).
\end{equation}
The angular matrix is
\begin{equation}
\label{eq:app-stage007-MAB20}
M_{AB}^{(20)}=\int_{S^2}Y_A Y_{20}Y_B\,d\Omega.
\end{equation}
Using the sixth moment of the unit sphere gives
\begin{equation}
\label{eq:app-stage007-axisym-matrix}
\boxed{
M^{(20)}=\frac{\sqrt5}{7\sqrt\pi}\operatorname{diag}\left(1,\frac12,\frac12,-1,-1\right)
}
\end{equation}
in the five real STF lanes.  Grouping the cosine/sine pairs gives
\begin{equation}
\label{eq:app-stage007-lambda-signature}
\boxed{
\lambda_{20}=1,
\qquad
\lambda_{21}=\frac12,
\qquad
\lambda_{22}=-1}.
\end{equation}
Any first-order grouped coefficient therefore has the form
\begin{equation}
\label{eq:app-stage007-x-signature}
x_{20}=x^{(0)}+\varepsilon x^{(1)},
\qquad
x_{21}=x^{(0)}+\frac{\varepsilon}{2}x^{(1)},
\qquad
x_{22}=x^{(0)}-\varepsilon x^{(1)}.
\end{equation}
The grouped trace/anomaly variables are
\begin{equation}
\label{eq:app-stage007-b-3a}
\boxed{
\bar x=x^{(0)},
\qquad
a_x=\frac{\varepsilon}{4}x^{(1)},
\qquad
b_x=\frac{3\varepsilon}{4}x^{(1)},
\qquad
b_x=3a_x}.
\end{equation}
This is the first weak-axisymmetric diagnostic line.

\paragraph{5. First-order normalization transport.}
If
\begin{equation}
D_A=D_0+\varepsilon\lambda_A D_1+O(\varepsilon^2),
\qquad
N_A=N_0+\varepsilon\lambda_A N_1+O(\varepsilon^2),
\end{equation}
then
\begin{equation}
\label{eq:app-stage007-PA-transport}
P_A=\frac{N_A}{D_A}=P_0+\varepsilon\lambda_A P_1+O(\varepsilon^2),
\qquad
\boxed{P_1=\frac{N_1D_0-N_0D_1}{D_0^2}}.
\end{equation}
The prefactor defects therefore satisfy
\begin{equation}
\label{eq:app-stage007-P-defects}
\boxed{
a_P=\frac{\varepsilon}{4}P_1,
\qquad
b_P=\frac{3\varepsilon}{4}P_1,
\qquad
b_P=3a_P}.
\end{equation}
