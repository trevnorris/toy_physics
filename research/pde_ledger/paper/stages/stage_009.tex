\subsection*{Derivation}

\paragraph{1. Axial basis.}
Use the normalized constant N/N profile
\begin{equation}
\label{eq:app-stage009-u0}
\nu_0(s)=\frac1{\sqrt L},
\qquad\nu_0'(0)=\nu_0'(L)=0.
\end{equation}
The D/N ladder is
\begin{equation}
\label{eq:app-stage009-DN-ladder}
f_n(s)=\sqrt{\frac2L}\sin\!\left(\frac{(n+1/2)\pi s}{L}\right),
\qquad
f_n(0)=0,
\qquad
f_n'(L)=0,
\end{equation}
with
\begin{equation}
\label{eq:app-stage009-kn}
k_n=\frac{(n+1/2)\pi}{L}.
\end{equation}
The minimal branch keeps \(n=0\).  The exact overlap is
\begin{equation}
\label{eq:app-stage009-kappa-n}
\kappa_n=\int_0^L \nu_0(s)f_n(s)\,ds
=\frac{\sqrt2}{(n+1/2)\pi},
\end{equation}
so
\begin{equation}
\label{eq:app-stage009-kappa0}
\boxed{
\kappa:=\kappa_0=\frac{2\sqrt2}{\pi}}.
\end{equation}

\paragraph{2. Branch-level coupling constants.}
Take
\begin{equation}
\label{eq:app-stage009-profile-choice}
\chi_\eta=\nu_0,
\qquad
\phi=f_0,
\qquad
u=\nu_0,
\qquad
w=f_0.
\end{equation}
Then
\begin{equation}
\label{eq:app-stage009-overlaps}
I_{\eta\phi}=\kappa,
\qquad
I_{\eta u}=1,
\qquad
I_{\eta w}=\kappa,
\qquad
I_{uw}=\kappa.
\end{equation}
Thus
\begin{equation}
\label{eq:app-stage009-couplings}
\boxed{
C=\kappa\lambda_B,
\qquad
G_U=\lambda_U,
\qquad
G_W=\kappa\lambda_W,
\qquad
R=\kappa\lambda_R}.
\end{equation}
Substituting into Stage~008 gives
\begin{equation}
\label{eq:app-stage009-DeltaQP}
\boxed{
\begin{aligned}
\Delta&=\Omega_U^2\Omega_W^2-\kappa^2\lambda_R^2,\\
Q&=\lambda_U^2\Omega_W^2+2\kappa^2\lambda_U\lambda_W\lambda_R+\kappa^2\lambda_W^2\Omega_U^2,\\
\mathcal P&=\kappa(\Omega_U^2\lambda_W+\lambda_R\lambda_U).
\end{aligned}
}
\end{equation}
The BdG softening is
\begin{equation}
\label{eq:app-stage009-X}
X=\frac{\kappa^2\lambda_B^2}{\varpi^2}.
\end{equation}

\paragraph{3. First finite-throat target and stiffness form.}
The target becomes
\begin{equation}
\label{eq:app-stage009-normalization}
\boxed{
\widehat m_{\rm rad}^{\,2}
\frac{\kappa^2(\Omega_U^2\lambda_W+\lambda_R\lambda_U)^2}
{\Delta\left(K\Delta-\Delta\kappa^2\lambda_B^2/\varpi^2-Q\right)}
=N_Q^{\rm target}}.
\end{equation}
Solving for the required wall stiffness gives
\begin{equation}
\label{eq:app-stage009-Kreq}
\boxed{
K_{\rm req}
=\frac{\kappa^2\lambda_B^2}{\varpi^2}
+\frac{Q}{\Delta}
+\frac{\widehat m_{\rm rad}^{\,2}\kappa^2(\Omega_U^2\lambda_W+\lambda_R\lambda_U)^2}
{N_Q^{\rm target}\Delta^2}}.
\end{equation}
For the constant wall profile,
\begin{equation}
\label{eq:app-stage009-Kgeo}
K=K_\eta+6T_\Omega.
\end{equation}
Thus the first concrete branch condition is
\begin{equation}
\label{eq:app-stage009-branch-condition}
K_\eta+6T_\Omega=K_{\rm req}.
\end{equation}

\stagefield{Output}{Stage~009 outputs the finite-throat D/N overlap \eqref{eq:app-stage009-kappa0}, the concrete couplings \eqref{eq:app-stage009-couplings}, the finite-throat normalization law \eqref{eq:app-stage009-normalization}, and the required stiffness formula \eqref{eq:app-stage009-Kreq}.}

\stagefield{Checks}{\begin{verificationchecklist}
\item Direct integration of \(\nu_0f_n\) gives \(\kappa_n=\sqrt2/[(n+1/2)\pi]\).
\item The lowest D/N mode is dominant because \(|\kappa_n|\) decreases as \((n+1/2)^{-1}\).
\item The support/mixed coupling vanishes if \(\kappa=0\), but the D/N lowest mode has \(\kappa>0\).
\item Solving \eqref{eq:app-stage009-normalization} for \(K\) gives \eqref{eq:app-stage009-Kreq}.
\end{verificationchecklist}}
