\subsection*{Derivation}

\paragraph{1. Static selected eigenvalue.}
Set
\begin{equation}
\label{eq:app-stage013-AB}
A=K_0-\Xi_0,
\qquad
B=K_1-\Xi_0=A+\Delta K_{\rm ax}.
\end{equation}
Define
\begin{equation}
\delta_\kappa=\kappa_0^2-\kappa_1^2,
\qquad
\Pi_\kappa=\kappa_0^2\kappa_1^2,
\qquad
\sigma=\kappa_0^2+\kappa_1^2.
\end{equation}
The lower eigenvalue is
\begin{equation}
\label{eq:app-stage013-lambda-minus}
\boxed{
\lambda_-=\frac{A+B-\alpha_0\sigma-
\sqrt{(\Delta K_{\rm ax}+\alpha_0\delta_\kappa)^2+4\alpha_0^2\Pi_\kappa}}{2}}.
\end{equation}
Let
\begin{equation}
R_\lambda=\sqrt{(\Delta K_{\rm ax}+\alpha_0\delta_\kappa)^2+4\alpha_0^2\Pi_\kappa}.
\end{equation}
The selected overlap is
\begin{equation}
\label{eq:app-stage013-s-minus}
\boxed{
s_-=(v\cdot e_-)^2
=\frac12\left[
\sigma+\frac{(\Delta K_{\rm ax}+\alpha_0\delta_\kappa)\delta_\kappa+4\alpha_0\Pi_\kappa}{R_\lambda}
\right]}.
\end{equation}
Equivalently, by Hellmann--Feynman,
\begin{equation}
\label{eq:app-stage013-HF}
\boxed{
s_-=-\frac{d\lambda_-}{d\alpha_0}}.
\end{equation}

\paragraph{2. Normalized selected response.}
Write
\begin{equation}
D_-(\omega)=D_{-0}+D_{-2}\omega^2+D_{-4}\omega^4-iC_{5,-}\omega^5+O(\omega^6),
\qquad
D_{-0}=\lambda_-.
\end{equation}
Then
\begin{equation}
Y_-(\omega)=\frac{D_{-0}}{D_-(\omega)}
=1+u_{2,-}\omega^2+u_{4,-}\omega^4+i\Gamma_{5,-}\omega^5+O(\omega^6),
\end{equation}
with
\begin{equation}
\label{eq:app-stage013-selected-even}
\boxed{u_{2,-}=-\frac{D_{-2}}{D_{-0}},
\qquad
u_{4,-}=\frac{D_{-2}^2-D_{-0}D_{-4}}{D_{-0}^2},
\qquad
\Gamma_{5,-}=\frac{C_{5,-}}{D_{-0}}}.
\end{equation}
The Stage~012 odd coefficient gives
\begin{equation}
\label{eq:app-stage013-Gamma-selected}
\Gamma_{5,-}=\frac{a^5}{27c_s^5}\frac{\beta_0s_-}{\lambda_-}.
\end{equation}
Therefore define
\begin{equation}
\label{eq:app-stage013-P0minus}
\boxed{
P_{0,-}=\frac{\beta_0s_-}{\lambda_-}
=-\beta_0\frac{d\ln\lambda_-}{d\alpha_0}}.
\end{equation}
The selected-branch target is
\begin{equation}
\label{eq:app-stage013-selected-target}
\boxed{
\widehat m_-^{\,2}P_{0,-}=N_Q^{\rm target}}.
\end{equation}

\stagefield{Output}{Stage~013 outputs the lower eigenvalue \eqref{eq:app-stage013-lambda-minus}, the selected overlap \eqref{eq:app-stage013-s-minus}, the Hellmann--Feynman identity \eqref{eq:app-stage013-HF}, the selected prefactor \eqref{eq:app-stage013-P0minus}, and the target \eqref{eq:app-stage013-selected-target}.}
