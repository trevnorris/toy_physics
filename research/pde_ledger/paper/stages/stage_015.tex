\subsection*{Derivation}

\paragraph{1. Natural D/N source coupling.}
Use
\begin{equation}
\label{eq:app-stage015-source-L}
L_{\rm src}=g_QQ_{\rm STF}\int_0^L\eta(s)f_0(s)\,ds.
\end{equation}
For \(\eta=q_0\nu_0+q_1\nu_1\), this gives source vector proportional to
\begin{equation}
v=(\kappa_0,\kappa_1)^T.
\end{equation}
Projection on the selected eigenvector gives
\begin{equation}
\label{eq:app-stage015-Jminus}
J_-=g_QQ_{\rm STF}(v\cdot e_-).
\end{equation}
At zero loading, the selected mode is the constant branch and
\begin{equation}
J_-(0)=g_QQ_{\rm STF}\kappa_0.
\end{equation}
Thus the selected source-map factor is
\begin{equation}
\label{eq:app-stage015-mhat-minus}
\boxed{
\widehat m_-=\frac{v\cdot e_-}{\kappa_0},
\qquad
\widehat m_-^{\,2}=\frac{s_-}{\kappa_0^2}}.
\end{equation}
Since \(s_-\) increases from \(\kappa_0^2\) toward \(\sigma=\kappa_0^2+\kappa_1^2\),
\begin{equation}
\label{eq:app-stage015-mhat-bound}
\boxed{
1\leq\widehat m_-^{\,2}<\frac{\sigma}{\kappa_0^2}=\frac{11}{9}}.
\end{equation}
The natural source-map factor is modest and cannot absorb arbitrary normalization error.

\paragraph{2. Source-map-corrected selected product.}
Combining \(P_{0,-}=\beta_0s_-/\lambda_-\) with \eqref{eq:app-stage015-mhat-minus} gives
\begin{equation}
\label{eq:app-stage015-Nminus}
\boxed{
N_-(\alpha_0):=\widehat m_-^{\,2}P_{0,-}
=\frac{\beta_0s_-^2}{\kappa_0^2\lambda_-}}.
\end{equation}
The selected target is
\begin{equation}
\label{eq:app-stage015-Nminus-target}
\boxed{
N_-(\alpha_0)=N_Q^{\rm target}}.
\end{equation}

\stagefield{Output}{Stage~015 outputs the source-map factor \eqref{eq:app-stage015-mhat-minus}, the bound \eqref{eq:app-stage015-mhat-bound}, and the invariant selected product \eqref{eq:app-stage015-Nminus}.}

\stagefield{Checks}{\begin{verificationchecklist}
\item The source vector is proportional to the same overlap vector \(v\) because the source uses the same D/N load profile.
\item At zero loading the normalization denominator is \(\kappa_0\), giving \(\widehat m_-=1\) at onset.
\item The upper bound \(11/9\) follows from \(\sigma/\kappa_0^2=(\kappa_0^2+\kappa_1^2)/\kappa_0^2=1+2/9\).
\end{verificationchecklist}}
