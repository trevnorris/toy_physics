\subsection*{Derivation}

Define the dimensionless continuum ratios
\begin{equation}
\label{eq:app-stage021-ratios-a}
\epsilon_\eta=\frac{c_{\eta U}^2}{K_UK_\eta^{\rm eff}},
\qquad
\epsilon_W=\frac{c_{UW}^2\sigma}{K_UK_W^{\rm eff}},
\qquad
\rho=\frac{c_{UW}c_{\eta U}}{K_Uc_{\eta W}},
\end{equation}
\begin{equation}
\label{eq:app-stage021-ratios-b}
Z_W=\frac{c_{\eta W}^2}{K_\eta^{\rm eff}K_W^{\rm eff}},
\qquad
\delta_0=\frac{\pi^2T_w}{L^2K_\eta^{\rm eff}},
\qquad
\Lambda=\frac{27\pi^2Gc_s^5K_W^{\rm eff}}{20a^5c^5\mu_W}.
\end{equation}
Substitution into Stage~020 gives the exact placement map
\begin{equation}
\label{eq:app-stage021-placement-map}
\boxed{
\delta=\frac{\delta_0}{1-\epsilon_\eta},
\qquad
M_{\rm mix}=\frac{8Z_W(1+\rho)^2}{\pi^2(1-\epsilon_\eta)(1-\epsilon_W)}},
\end{equation}
\begin{equation}
\label{eq:app-stage021-Rtarget}
\boxed{
R_{\rm target}=\frac{\Lambda(1-\epsilon_\eta)(1-\epsilon_W)^2}{Z_W(1+\rho)^2}}.
\end{equation}
Multiplying \eqref{eq:app-stage021-placement-map} and \eqref{eq:app-stage021-Rtarget} cancels the redistribution variables:
\begin{equation}
\label{eq:app-stage021-product-law}
\boxed{
R_{\rm target}M_{\rm mix}
=\frac{8\Lambda(1-\epsilon_W)}{\pi^2}
=\frac{54Gc_s^5K_W^{\rm eff}(1-\epsilon_W)}{5a^5c^5\mu_W}}.
\end{equation}
Thus \((\epsilon_W,\Lambda)\) fix the product scale, \(\epsilon_\eta\) fixes the geometry lane through \(\delta\), and \((\epsilon_\eta,Z_W,\rho)\) redistribute the point along a fixed product curve.
