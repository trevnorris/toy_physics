\paragraph{Mixed gauge invariants.}
Under \(A_0\mapsto A_0-\partial_t\chi\), \(A_a\mapsto A_a+\partial_a\chi\),
and \(A_w\mapsto A_w+\partial_w\chi\), the mixed fields
\begin{equation}
\label{eq:app-stage021-mixed-fields}
E_w=-\partial_tA_w-\partial_wA_0,
\qquad
C_a=\partial_aA_w-\partial_wA_a
\end{equation}
are gauge invariant.

\paragraph{Reduced lane.}
For one harmonic lane, eliminating the brane-like coordinate \(A_l\) and the
mixed coordinate \(W_l\) gives
\begin{equation}
\label{eq:app-stage021-self-energy}
\Sigma_l^{({\rm EM}+{\rm mix},{\rm cons})}(\omega)
=
\frac{g_{A,l}^2W_l(\omega)+2g_{A,l}g_{W,l}R_l+g_{W,l}^2A_l(\omega)}
{A_l(\omega)W_l(\omega)-R_l^2}.
\end{equation}
The low-frequency conservative coefficients are
\begin{align}
\label{eq:app-stage021-z-coeffs}
z_0&=\frac{N_0^{\rm cons}}{\Delta_0},\\
z_2&=\frac{N_0^{\rm cons}S_2-G_2\Delta_0}{\Delta_0^2},\\
z_4&=\frac{N_0^{\rm cons}(S_2^2-\Delta_0)-S_2G_2\Delta_0}{\Delta_0^3}.
\end{align}

\paragraph{Outgoing transfer.}
Replacing \(W_l(\omega)\) by \(W_l(\omega)-\Pi_l^{\rm out}(\omega)\) gives
the first-order transfer factor
\begin{equation}
\label{eq:app-stage021-transfer-factor}
N_l(\omega)=
\frac{\bigl[A_l(\omega)g_{W,l}+R_lg_{A,l}\bigr]^2}
{\bigl[A_l(\omega)W_l(\omega)-R_l^2\bigr]^2}.
\end{equation}
In particular,
\begin{equation}
\label{eq:app-stage021-n0-positive}
N_l(0)=
\frac{(\Omega_{A,l}^2g_{W,l}+R_lg_{A,l})^2}
{(\Omega_{A,l}^2\Omega_{W,l}^2-R_l^2)^2}\ge0.
\end{equation}

\paragraph{Compact outgoing fingerprint.}
For the compact outgoing \(l=2\) branch,
\begin{equation}
\label{eq:app-stage021-outgoing-fingerprint}
\widehat Y_2^{\rm out}(\omega)
=1+\frac{a^2\omega^2}{9c_s^2}
+\frac{4a^4\omega^4}{81c_s^4}
+i\frac{a^5\omega^5}{27c_s^5}
+O(\omega^6),
\end{equation}
so
\begin{equation}
\label{eq:app-stage021-wall-odd}
\delta D_2^{\rm odd}(\omega)
=-iN_2(0)\frac{a^5}{27c_s^5}\omega^5+O(\omega^7).
\end{equation}
