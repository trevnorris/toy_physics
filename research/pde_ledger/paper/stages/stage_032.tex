\subsection*{Derivation}

The physical coherent support ratio is
\begin{equation}
\label{eq:app-stage032-zeta-phys}
\boxed{
\zeta_n^{\rm phys}
=\frac{K_W^{\rm eff}}{K_{\phi,n}^{\rm eff}}
\left(\frac{I_n}{I_0}\right)^2}.
\end{equation}
For a uniform local source density,
\begin{equation}
\label{eq:app-stage032-In-ratio}
\boxed{
\frac{I_n}{I_0}=\frac{1}{2n+1}}.
\end{equation}
For a same-operator twin family,
\begin{equation}
\label{eq:app-stage032-zeta-twin}
\boxed{
\zeta_n^{\rm twin}=\frac{1}{(2n+1)^2[1+xn(n+1)]},
\qquad
x=\frac{\pi^2T_X}{L^2K_W^{\rm eff}}}.
\end{equation}
The lowest twin value is
\begin{equation}
\label{eq:app-stage032-lowest-twin}
\boxed{
\zeta_0^{\rm twin}=1}.
\end{equation}

\stagefield{Output}{Microscopic D/N support ratios \eqref{eq:app-stage032-zeta-phys}--\eqref{eq:app-stage032-zeta-twin}.}
\stagefield{Downstream use}{Stage~033 converts these ratios into explicit selection rules.}
