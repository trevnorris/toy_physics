\stagefield{Purpose}{Stage~034 eliminates the selected eigenvector variables by introducing a scalar softening depth \(x=A-\lambda_-\).  In this variable, the total loading, selected source overlap, invariant normalization product, and required support contribution all become explicit rational functions.}

\stagefield{Inputs}{The inputs are the rank-one two-level secular equation and the selected product \(N_-\) from Stage~032.}

\subsection*{Derivation}

\paragraph{1. Softening-depth variable and secular equation.}
Define
\begin{equation}
\label{eq:app-stage034-x}
\boxed{x:=A-\lambda_-}.
\end{equation}
On the stable branch,
\begin{equation}
0\leq x<A,
\qquad
\lambda_-=A-x.
\end{equation}
The rank-one secular equation is
\begin{equation}
\label{eq:app-stage034-secular}
1=\alpha_0\left(\frac{\kappa_0^2}{x}+\frac{\kappa_1^2}{x+\Delta K_{\rm ax}}\right).
\end{equation}
Solving for total loading gives
\begin{equation}
\label{eq:app-stage034-alpha-x}
\boxed{
\alpha_0(x)=
\frac{x(x+\Delta K_{\rm ax})}{\kappa_0^2(x+\Delta K_{\rm ax})+\kappa_1^2x}}.
\end{equation}

\paragraph{2. Selected overlap and invariant product.}
The selected overlap can be written as \(s_-=dx/d\alpha_0\), giving
\begin{equation}
\label{eq:app-stage034-s-x}
\boxed{
s_-(x)=
\frac{\left[\kappa_0^2(x+\Delta K_{\rm ax})+\kappa_1^2x\right]^2}
{\kappa_0^2(x+\Delta K_{\rm ax})^2+\kappa_1^2x^2}}.
\end{equation}
The invariant selected product is
\begin{equation}
\label{eq:app-stage034-N-x}
\boxed{
N_-(x)=
\frac{\beta_0\left[\kappa_0^2(x+\Delta K_{\rm ax})+\kappa_1^2x\right]^4}
{\kappa_0^2(A-x)\left[\kappa_0^2(x+\Delta K_{\rm ax})^2+\kappa_1^2x^2\right]^2}}.
\end{equation}
The target is
\begin{equation}
\label{eq:app-stage034-N-target}
\boxed{
N_-(x)=N_Q^{\rm target},
\qquad
0\leq x<A}.
\end{equation}

\paragraph{3. Monotone loading and required support.}
Differentiating \eqref{eq:app-stage034-alpha-x} gives
\begin{equation}
\label{eq:app-stage034-dalpha-dx}
\boxed{
\frac{d\alpha_0}{dx}=
\frac{\kappa_0^2(x+\Delta K_{\rm ax})^2+\kappa_1^2x^2}
{\left[\kappa_0^2(x+\Delta K_{\rm ax})+\kappa_1^2x\right]^2}>0}.
\end{equation}
Thus \(x\) is a one-to-one stable branch coordinate.  Splitting
\begin{equation}
\alpha_0=\frac{g_B^2}{\varpi^2}+\alpha_{\rm mix},
\qquad
\alpha_{\rm mix}=\frac{\mathcal X^2}{\Omega_U^2\Delta_0},
\end{equation}
the required support loading is
\begin{equation}
\label{eq:app-stage034-support-req}
\boxed{
\frac{g_{B,{\rm req}}^2}{\varpi^2}
=
\frac{x(x+\Delta K_{\rm ax})}{\kappa_0^2(x+\Delta K_{\rm ax})+\kappa_1^2x}
-
\frac{\mathcal X^2}{\Omega_U^2\Delta_0}}.
\end{equation}
The nonnegativity of this expression is the support-feasibility condition.

\stagefield{Output}{Stage~034 outputs the softening-depth variable \eqref{eq:app-stage034-x}, the loading law \eqref{eq:app-stage034-alpha-x}, the selected overlap \eqref{eq:app-stage034-s-x}, the normalization product \eqref{eq:app-stage034-N-x}, and the required support formula \eqref{eq:app-stage034-support-req}.}

\stagefield{Checks}{\begin{verificationchecklist}
\item The secular equation follows from the rank-one determinant lemma.
\item The identity \(s_-=dx/d\alpha_0\) follows because \(x=A-\lambda_-\) and \(s_-=-d\lambda_-/d\alpha_0\).
\item \(d\alpha_0/dx>0\) proves that \(x\) is a good branch coordinate.
\item The support requirement is obtained by subtracting the mixed baseline from the required total load.
\end{verificationchecklist}}
