\subsection*{Derivation}

Define
\begin{equation}
\label{eq:app-stage034-Cmix}
C_{\rm mix}=\frac{8\Lambda(1-\epsilon)}{\pi^2}.
\end{equation}
The required enhancement is \(S_{\rm req}=\Pi_{\rm tr}/C_{\rm mix}\).  Since the lowest twin gives \(S_0=2\), it is sufficient iff
\begin{equation}
\label{eq:app-stage034-twin-criterion}
\boxed{
\Pi_{\rm tr}\leq2C_{\rm mix}
=\frac{16\Lambda(1-\epsilon)}{\pi^2}}.
\end{equation}
Equivalently, the threshold scales can be written as
\begin{equation}
\label{eq:app-stage034-Lambda-threshold}
\Lambda_{\rm twin,req}=\frac{\pi^2}{16(1-\epsilon)}\Pi_{\rm tr},
\qquad
M_{\rm mix}^{({\rm twin,req})}=\frac{G_{\rm tr}}{2}.
\end{equation}

\stagefield{Output}{The lowest-twin branch-product criterion \eqref{eq:app-stage034-twin-criterion}.}
\stagefield{Downstream use}{Stage~035 uses this criterion to split the support problem into mixed-only, lowest-twin, and non-twin regimes.}
