\stagefield{Purpose}{Stage~035 non-dimensionalizes the softening-depth target.  The selected normalization product factors into an onset value times a universal D/N shape function \(F(\xi,\delta)\).  Strict monotonicity of \(F\) upgrades the Stage~033 onset inequality into a unique stable normalization locus.}

\stagefield{Inputs}{The inputs are the Stage~034 expression for \(N_-(x)\) and the exact D/N constants \(\kappa_0^2=8/\pi^2\), \(\kappa_1^2=16/(9\pi^2)\).}

\subsection*{Derivation}

\paragraph{1. Dimensionless variables.}
Use
\begin{equation}
\label{eq:app-stage035-DN-constants}
\boxed{
\kappa_0^2=\frac{8}{\pi^2},
\qquad
\kappa_1^2=\frac{16}{9\pi^2},
\qquad
\eta=\frac{\kappa_1^2}{\kappa_0^2}=\frac29}.
\end{equation}
Define
\begin{equation}
\label{eq:app-stage035-xi-delta}
\boxed{
\xi=\frac{x}{A},
\qquad
\delta=\frac{\Delta K_{\rm ax}}{A},
\qquad
0\leq\xi<1}.
\end{equation}
Then
\begin{equation}
\label{eq:app-stage035-factorization}
N_-(x)=N_-(0)F(\xi,\delta),
\qquad
N_-(0)=\frac{\beta_0\kappa_0^2}{A}.
\end{equation}
The exact shape function is
\begin{equation}
\label{eq:app-stage035-F}
\boxed{
F(\xi,\delta)=
\frac{(9\delta+11\xi)^4}{81(1-\xi)(9\delta^2+18\delta\xi+11\xi^2)^2}}.
\end{equation}
The dimensionless target ratio is
\begin{equation}
\label{eq:app-stage035-Rtarget}
\boxed{
R_{\rm target}:=\frac{N_Q^{\rm target}}{N_-(0)}
=\frac{N_Q^{\rm target}A}{\beta_0\kappa_0^2}}.
\end{equation}
The normalization locus is
\begin{equation}
\label{eq:app-stage035-F-locus}
\boxed{F(\xi,\delta)=R_{\rm target}}.
\end{equation}

\paragraph{2. Monotonicity and endpoints.}
Differentiating gives
\begin{equation}
\label{eq:app-stage035-F-derivative}
\boxed{
\frac{\partial F}{\partial\xi}
=
\frac{(9\delta+11\xi)^3
\left(81\delta^3+72\delta^2+206\delta^2\xi+297\delta\xi^2+138\xi^3\right)}
{81(1-\xi)^2(9\delta^2+18\delta\xi+11\xi^2)^3}>0}.
\end{equation}
Moreover,
\begin{equation}
\label{eq:app-stage035-F-endpoints}
F(0,\delta)=1,
\qquad
\lim_{\xi\to1^-}F(\xi,\delta)=+\infty.
\end{equation}
Near softening,
\begin{equation}
\label{eq:app-stage035-Cdelta}
F(\xi,\delta)\sim\frac{C(\delta)}{1-\xi},
\qquad
C(\delta)=\frac{(9\delta+11)^4}{81(9\delta^2+18\delta+11)^2}.
\end{equation}
Therefore
\begin{equation}
\label{eq:app-stage035-existence}
\boxed{
R_{\rm target}<1\Rightarrow\text{no stable hit},
\qquad
R_{\rm target}=1\Rightarrow\xi_{\rm req}=0,
\qquad
R_{\rm target}>1\Rightarrow\exists!\,\xi_{\rm req}\in(0,1)}.
\end{equation}

\paragraph{3. Required total loading.}
The total loading law becomes
\begin{equation}
\label{eq:app-stage035-alpha-req}
\boxed{
\alpha_{\rm req}(\xi,\delta)=
\frac{9\pi^2A\xi(\xi+\delta)}{8(9\delta+11\xi)}}.
\end{equation}
As \(\xi\to1^-\), this approaches the refined stable threshold
\begin{equation}
\label{eq:app-stage035-alpha-crit}
\alpha_{\rm crit}=\frac{9\pi^2A(1+\delta)}{8(11+9\delta)}.
\end{equation}

\paragraph{4. Useful asymptotics.}
Near onset,
\begin{equation}
\label{eq:app-stage035-onset-expansion}
F(\xi,\delta)=1+\left(1+\frac{8}{9\delta}\right)\xi
+\left(1+\frac{8}{9\delta}-\frac{28}{27\delta^2}\right)\xi^2+O(\xi^3).
\end{equation}
If \(R_{\rm target}=1+\varepsilon_R\), then
\begin{equation}
\label{eq:app-stage035-xi-near-onset}
\xi_{\rm req}\simeq\frac{\varepsilon_R}{1+8/(9\delta)}.
\end{equation}
For large target ratio,
\begin{equation}
\label{eq:app-stage035-xi-near-softening}
1-\xi_{\rm req}\simeq\frac{C(\delta)}{R_{\rm target}}.
\end{equation}

\stagefield{Output}{Stage~035 outputs the shape function \eqref{eq:app-stage035-F}, the target ratio \eqref{eq:app-stage035-Rtarget}, the monotonicity derivative \eqref{eq:app-stage035-F-derivative}, the existence/uniqueness theorem \eqref{eq:app-stage035-existence}, and the required total loading \eqref{eq:app-stage035-alpha-req}.}
