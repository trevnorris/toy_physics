\subsection*{Derivation}

The quadratic support energy is
\begin{equation}
\label{eq:app-stage053-E2}
E^{(2)}[q]=\frac12\int_0^L ds\,[T_Xq'(s)^2+K_Xq(s)^2].
\end{equation}
The coefficients are
\begin{equation}
\label{eq:app-stage053-TX-KX}
\boxed{
T_X=\frac{\hbar^2}{4m\rho_w}N_{\phi\phi},
\qquad
K_X=H_wN_{\phi\phi}+\frac{\hbar^2}{4m\rho_w}G_{\phi\phi},
\qquad
H_w=\frac{mc_{s,w}^2}{\rho_w}}.
\end{equation}
For a thin shell with moments \(I_f=\int f'^2d\xi\) and \(I_g=\int f''^2d\xi\),
\begin{equation}
\label{eq:app-stage053-thin-TK}
\boxed{
T_X=\frac{\pi a^2\ell I_f\hbar^2}{m\rho_w},
\qquad
K_X=4\pi a^2\ell I_fH_w+\frac{\pi a^2I_g\hbar^2}{m\rho_w\ell}}.
\end{equation}
Thus
\begin{equation}
\label{eq:app-stage053-kappa}
\boxed{
\kappa=\frac{K_XL^2}{T_X}
=4\left(\frac{mc_{s,w}L}{\hbar}\right)^2+\frac{I_g}{I_f}\left(\frac{L}{\ell}\right)^2}.
\end{equation}
The matched wall control is
\begin{equation}
\label{eq:app-stage053-Wwall-Xi}
\boxed{
W_{\rm wall}=\Xi=\frac{4\rho_w^2V_0^2L^2}{\hbar^2c_{s,w}^2\ell^2}}.
\end{equation}
