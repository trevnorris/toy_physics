\label{eq:app-stage054-tanh}
f(\xi)=\frac{1+\tanh\xi}{2},
\qquad
f'(\xi)=\frac12\operatorname{sech}^2\xi,
\end{equation}

\begin{equation}
\label{eq:app-stage054-moments}
\boxed{
I_f=\frac13,
\qquad
I_g=\frac{4}{15},
\qquad
\frac{I_g}{I_f}=\frac45}.
\end{equation}
The natural local mouth closure is
\begin{equation}
\label{eq:app-stage054-mouth}
K_m=\frac{T_X}{\ell},
\qquad
\eta=\frac{K_mL}{T_X}=\frac{L}{\ell}=\Lambda_\ell.
\end{equation}
Define
\begin{equation}
\label{eq:app-stage054-controls}
\chi_s=\frac{mc_{s,w}L}{\hbar},
\qquad
\Lambda_\ell=\frac{L}{\ell},
\qquad
\Upsilon_w=\frac{4\rho_w^2V_0^2}{\hbar^2c_{s,w}^2}.
\end{equation}
Then
\begin{equation}
\label{eq:app-stage054-control-laws}
\boxed{
\kappa=4\chi_s^2+\frac45\Lambda_\ell^2,
\qquad
\eta=\Lambda_\ell,
\qquad
W_{\rm wall}=\Upsilon_w\Lambda_\ell^2}.
\end{equation}
