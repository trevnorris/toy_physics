\subsection*{Derivation}

Take
\begin{equation}
\label{eq:app-stage062-dV}
\delta V_{\rm conf}(s,y)=-g_\phi\chi_\phi(y)\phi(s).
\end{equation}
For the frozen \(n=5\) EOS,
\begin{equation}
\label{eq:app-stage062-hprime}
h'(\rho_*)=5K\rho_*^3=\frac{mc_{s,*}^2}{\rho_*}.
\end{equation}
Define profile overlaps
\begin{equation}
\label{eq:app-stage062-overlaps}
N_{\sigma\sigma}=\int d^3y\,\chi_\sigma^2,
\qquad
N_{\phi\phi}=\int d^3y\,\chi_\phi^2,
\qquad
O_{\sigma\phi}=\int d^3y\,\chi_\sigma\chi_\phi.
\end{equation}
Then
\begin{equation}
\label{eq:app-stage062-Gmicro}
\boxed{
G_{\rm micro}
=\frac{\rho_*g_\phi^2O_{\sigma\phi}^2}{mc_{s,*}^2K_XN_{\sigma\sigma}}
=\frac{\rho_*g_\phi^2N_{\phi\phi}}{mc_{s,*}^2K_X}C_{\sigma\phi}^2},
\end{equation}
where
\begin{equation}
\label{eq:app-stage062-Csigma}
C_{\sigma\phi}^2=\frac{O_{\sigma\phi}^2}{N_{\sigma\sigma}N_{\phi\phi}},
\qquad
0\leq C_{\sigma\phi}^2\leq1.
\end{equation}

\stagefield{Output}{Parent-action gain formula \eqref{eq:app-stage062-Gmicro}.}
\stagefield{Downstream use}{Stage~063 converts this into amplitude/coherence thresholds; Stage~064 derives the equilibrium source profile.}
