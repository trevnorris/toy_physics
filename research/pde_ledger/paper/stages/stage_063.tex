\subsection*{Derivation}

At fixed profile shapes, the amplitude thresholds are
\begin{equation}
\label{eq:app-stage063-gphi-thresholds}
\boxed{
g_{\phi,{\rm fail}}^2
=\frac{mc_{s,*}^2K_XN_{\sigma\sigma}G_{\rm fail}}{\rho_*O_{\sigma\phi}^2},
\qquad
 g_{\phi,{\rm suff}}^2
=\frac{mc_{s,*}^2K_XN_{\sigma\sigma}G_{\rm suff}}{\rho_*O_{\sigma\phi}^2}}.
\end{equation}
At fixed \(g_\phi\), the coherence thresholds are
\begin{equation}
\label{eq:app-stage063-coherence-thresholds}
\boxed{
C_{\rm fail}^2=\frac{mc_{s,*}^2K_XG_{\rm fail}}{\rho_*g_\phi^2N_{\phi\phi}},
\qquad
C_{\rm suff}^2=\frac{mc_{s,*}^2K_XG_{\rm suff}}{\rho_*g_\phi^2N_{\phi\phi}}}.
\end{equation}
The Cauchy bound yields the no-go:
\begin{equation}
\label{eq:app-stage063-cauchy-nogo}
C_{\rm fail}^2>1
\quad\Longrightarrow\quad
\text{no source profile in this support channel can reach even the fail threshold.}
\end{equation}
