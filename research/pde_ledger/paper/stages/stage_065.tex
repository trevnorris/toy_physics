\subsection*{Derivation}

The demand is
\begin{equation}
\label{eq:app-stage065-zeta-req}
\zeta_{\rm req}(\Pi_{\rm tr},C_{\rm mix},\epsilon_{\rm blk})
=\frac{\Pi_{\rm tr}-C_{\rm mix}}
{C_{\rm mix}-\epsilon_{\rm blk}(2C_{\rm mix}-\Pi_{\rm tr})}.
\end{equation}
The physical supply is
\begin{equation}
\label{eq:app-stage065-zeta-phys}
\zeta_{\rm phys}(\mathrm{Pe},\eta;\kappa)
=\Omega_{\mathrm{Pe}}^2\frac{\kappa+\pi^2/4}{\kappa+y(\eta)^2}.
\end{equation}
With \(\mathrm{Pe}_*\) selected by Stage~041, define
\begin{equation}
\label{eq:app-stage065-Rquad}
\boxed{
R_{\rm quad}
=\zeta_{\rm req}(\Pi_{\rm tr},C_{\rm mix},\epsilon_{\rm blk})
-\zeta_{\rm phys}(\mathrm{Pe}_*(\Xi,\eta,\kappa),\eta;\kappa)}.
\end{equation}
The sign convention is
\begin{equation}
\label{eq:app-stage065-sign}
R_{\rm quad}<0\Rightarrow\text{support exceeds demand},
\quad
R_{\rm quad}=0\Rightarrow\text{saturation},
\quad
R_{\rm quad}>0\Rightarrow\text{support failure}.
\end{equation}
The Family--1 specialization is obtained by setting \((\eta,\kappa)=(37,12321/5)\) and
\begin{equation}
\label{eq:app-stage065-XiF1}
\Xi_{\rm F1}=W_{\rm wall}=1369\Upsilon_w=136900\Theta_w.
\end{equation}

\stagefield{Output}{The master residual \eqref{eq:app-stage065-Rquad}.}
