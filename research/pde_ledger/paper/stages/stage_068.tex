\subsection*{Derivation}

The selected product and mixed product are
\begin{equation}
\label{eq:app-stage068-products}
\boxed{
\Pi_{\rm tr}=\frac{N_Q^{\rm target}}{\beta_0}\alpha_{\rm req},
\qquad
C_{\rm mix}=\frac{N_Q^{\rm target}}{\beta_0}\alpha_{\rm mix}}.
\end{equation}
Therefore the absolute factors cancel:
\begin{equation}
\label{eq:app-stage068-ratio}
\boxed{
\frac{\Pi_{\rm tr}}{C_{\rm mix}}
=\frac{\alpha_{\rm req}}{\alpha_{\rm mix}}
=:\rho_\alpha}.
\end{equation}
The support demand becomes
\begin{equation}
\label{eq:app-stage068-zeta-rho}
\boxed{
\zeta_{\rm req}(\rho_\alpha,\epsilon_{\rm blk})
=\frac{\rho_\alpha-1}{1-\epsilon_{\rm blk}(2-\rho_\alpha)}}.
\end{equation}
In the unblocked limit, \(\zeta_{\rm req}=\rho_\alpha-1\).

\stagefield{Output}{The cancellation theorem \eqref{eq:app-stage068-ratio} and reduced demand formula \eqref{eq:app-stage068-zeta-rho}.}
\stagefield{Downstream use}{Stages~069--070 express the Family--1 support window entirely in \(\rho_\alpha\); Stage~071 computes \(\rho_\alpha\) from the minimal module.}
