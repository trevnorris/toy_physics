\subsection*{Derivation}

The minimal precursor is
\begin{equation}
\label{eq:app-stage071-YQ}
\boxed{
Y_Q^{\rm cons}(\omega)
=c_0+\frac{c_1}{1-\omega^2/\Omega_Q^2},
\qquad
c_0=\frac34,
\qquad
c_1=\frac14,
\qquad
\Omega_Q=\frac{3c_s}{2a}}.
\end{equation}
The natural contact-plus-pole reading writes
\begin{equation}
\label{eq:app-stage071-contact-pole}
Y_Q^{\rm cons}(\omega)
=\frac{1}{\rho_\alpha}
+\frac{\rho_\alpha-1}{\rho_\alpha}\frac{1}{1-\omega^2/\Omega_Q^2}.
\end{equation}
Matching coefficients gives
\begin{equation}
\label{eq:app-stage071-rho}
\boxed{
\rho_\alpha=\frac{1}{c_0}=\frac43,
\qquad
\zeta_{\rm req}=\frac{c_1}{c_0}=\frac13,
\qquad
\Pi_{\rm tr}=\frac43C_{\rm mix}}.
\end{equation}
This lies in the symmetric-lowest-twin regime:
\begin{equation}
\label{eq:app-stage071-regime}
C_{\rm mix}<\Pi_{\rm tr}<2C_{\rm mix},
\qquad
0<\zeta_{\rm req}=\frac13<1.
\end{equation}

\stagefield{Output}{The minimal-module ratio \eqref{eq:app-stage071-rho}.}
