\subsection*{Derivation}

Let \(y_{\rm F1}\) solve
\begin{equation}
\label{eq:app-stage079-yF1}
y_{\rm F1}\tan y_{\rm F1}=37,
\qquad
0<y_{\rm F1}<\frac{\pi}{2},
\qquad
 y_{\rm F1}\simeq1.52948248371470.
\end{equation}
Define
\begin{equation}
\label{eq:app-stage079-AF1}
A_{\rm F1}=\frac{\kappa_{\rm F1}+\pi^2/4}{\kappa_{\rm F1}+y_{\rm F1}^2},
\qquad
\kappa_{\rm F1}=\frac{12321}{5}.
\end{equation}
Then
\begin{equation}
\label{eq:app-stage079-AF1-value}
\boxed{
A_{\rm F1}\simeq1.00005192880220}.
\end{equation}
The Family--1 support-ratio map is
\begin{equation}
\label{eq:app-stage079-zetaF1}
\boxed{
\zeta_{\rm F1}(\mathrm{Pe})=A_{\rm F1}\Omega_{\mathrm{Pe}}^2}.
\end{equation}
The hard constructive ceiling is
\begin{equation}
\label{eq:app-stage079-ceiling}
\boxed{
\zeta_{\max}^{\rm F1}=A_{\rm F1}\frac{\pi^2}{4}
\simeq2.46752922945601}.
\end{equation}

\stagefield{Output}{Family--1 demand map \eqref{eq:app-stage079-zetaF1} and ceiling \eqref{eq:app-stage079-ceiling}.}
